   141	fn validate_patient(record: &Value, config: &ClinicalConfig) -> Result<()> {
   142	    if record["uuid"].as_str() != Some(config.patient_uuid.as_str()) {
   143	        return Err(anyhow!("Patient UUID mismatch"));
   144	    }
   145	
   146	    if record["voided"].as_bool() != Some(false) {

   148	    }

   153	fn validate_encounter(record: &Value, config: &ClinicalConfig) -> Result<()> {
   154	    if record["uuid"].as_str() != Some(config.encounter_uuid.as_str()) {
   155	        return Err(anyhow!("Encounter UUID mismatch"));
   156	    }
   157	
   158	    if record["patient"]["uuid"].as_str() != Some(config.patient_uuid.as_str()) {
   159	        return Err(anyhow!("Encounter patient UUID mismatch"));
   160	    }
   161	
   162	    if record["voided"].as_bool() != Some(false) {
   163	        return Err(anyhow!("Encounter is voided"));
   164	    }
   165	
   166	    let observations = record["obs"]
   167	        .as_array()
   168	        .ok_or_else(|| anyhow!("Encounter observations missing"))?;
   169	
   170	    let observation = observations
   171	        .iter()
   172	        .find(|obs| obs["uuid"].as_str() == Some(config.observation_uuid.as_str()))
   173	        .ok_or_else(|| anyhow!("Expected observation not found"))?;
   174	
   175	    if observation["voided"].as_bool() != Some(false) {
   176	        return Err(anyhow!("Observation is voided"));
   177	    }
   178	
   179	    if observation["concept"]["uuid"].as_str() != Some(config.concept_uuid.as_str()) {
   180	        return Err(anyhow!("Observation concept mismatch"));
   181	    }
   182	
   183	    let value = observation["value"]
   184	        .as_f64()
   185	        .ok_or_else(|| anyhow!("Observation value is not numeric"))?;
   186	
   187	    if (value - config.expected_temperature).abs() > 0.001 {
   188	        return Err(anyhow!(
   189	            "Temperature mismatch: expected {}, received {}",
   190	            config.expected_temperature,
   191	            value
   192	        ));
   193	    }
   194	
   195	    Ok(())
   196	}

   198	async fn clinical_request(
   199	    client: &Client,
   200	    config: &ClinicalConfig,
   201	    status: &mut u16,
   202	) -> Result<()> {
   203	    let base = config.base_url.trim_end_matches('/');
   204	
   205	    let patient_url = format!("{}/ws/rest/v1/patient/{}", base, config.patient_uuid);
   206	
   207	    let response = client
   208	        .get(patient_url)
   209	        .basic_auth(&config.username, Some(&config.password))
   210	        .send()
   211	        .await?;
   212	
   213	    *status = response.status().as_u16();
   214	
   215	    if !response.status().is_success() {
   216	        return Err(anyhow!("Patient HTTP {}", status));
   217	    }
   218	
   219	    let patient: Value = response.json().await?;
   220	    validate_patient(&patient, config)?;
   221	
   222	    let encounter_url = format!(
   223	        "{}/ws/rest/v1/encounter/{}?v=full",
   224	        base, config.encounter_uuid
   225	    );
   226	
   227	    let response = client
   228	        .get(encounter_url)
   229	        .basic_auth(&config.username, Some(&config.password))
   230	        .send()
   231	        .await?;
   232	
   233	    *status = response.status().as_u16();
   234	
   235	    if !response.status().is_success() {
   236	        return Err(anyhow!("Encounter HTTP {}", status));
   237	    }
   238	
   239	    let encounter: Value = response.json().await?;
   240	    validate_encounter(&encounter, config)?;
   241	
   242	    Ok(())
   243	}

   245	async fn run_clinical_probe(
   246	    client: Client,
   247	    config: ClinicalConfig,
   248	    output_path: String,
   249	    interval_ms: u64,
   250	) -> Result<()> {
   251	    let file = OpenOptions::new()
   252	        .create(true)
   253	        .append(true)
   254	        .open(output_path)?;
   255	
   256	    let mut writer = Writer::from_writer(file);
   257	
   258	    println!("[CLINICAL] Clinical-data probe started");
   259	
   260	    loop {
   261	        let timestamp = Utc::now();
   262	        let start = Instant::now();
   263	        let mut status = 0;
   264	
   265	        let result = clinical_request(&client, &config, &mut status).await;
   266	
   267	        let observation = ClinicalObservation {
   268	            timestamp_utc: timestamp.to_rfc3339(),
   269	            patient_uuid: config.patient_uuid.clone(),
   270	            encounter_uuid: config.encounter_uuid.clone(),
   271	            status,
   272	            latency_ms: start.elapsed().as_secs_f64() * 1000.0,
   273	            success: result.is_ok(),
   274	            error: result.err().map(|e| e.to_string()).unwrap_or_default(),
   275	        };
   276	
   277	        writer.serialize(&observation)?;
   278	        writer.flush()?;
   279	
   280	        println!(
   281	            "[CLINICAL] {} | status={} | latency={:.3}ms | success={} | error={}",
   282	            observation.timestamp_utc,
   283	            observation.status,
   284	            observation.latency_ms,
   285	            observation.success,
   286	            observation.error
   287	        );
   288	
   289	        sleep(Duration::from_millis(interval_ms)).await;
   290	    }
   291	}

   293	async fn run_standard_http_probe(
   294	    config: HarnessConfig,
   295	    output_path: String,
   296	    client: Client,
   297	) -> Result<()> {
   298	    let file = OpenOptions::new()
   299	        .create(true)
   300	        .append(true)
   301	        .open(output_path)?;
   302	    let mut writer = Writer::from_writer(file);
   303	
   304	    loop {
   305	        let timestamp = Utc::now();
   306	        let stable_url = config.measurement_url.as_deref().filter(|s| !s.is_empty());
   307	
   308	        let observation = if let Some(url) = stable_url {
   309	            // Start the HTTP request independently of Kubernetes discovery.
   310	            // Both futures run concurrently; an API timeout cannot suppress HTTP.
   311	            let http_request = async {
   312	                let started = Instant::now();
   313	                let result = client.get(url).send().await;
   314	                (result, started.elapsed().as_secs_f64() * 1000.0)
   315	            };
   316	            let ((http_result, latency_ms), placement_result) = tokio::join!(
   317	                http_request,
   318	                discover_active_node(&config.namespace, &config.pod_label_selector)
   319	            );
   320	            let (node, placement_status, placement_error) = match placement_result {
   321	                Ok(Some(node)) => (node, "KNOWN".to_string(), String::new()),
   322	                Ok(None) => (String::new(), "UNKNOWN".to_string(),
   323	                    "no Running workload Pod found".to_string()),
   324	                Err(error) => (String::new(), "UNKNOWN".to_string(), error.to_string()),
   325	            };
   326	            match http_result {
   327	                Ok(response) => HttpObservation {
   328	                    timestamp_utc: timestamp.to_rfc3339(),
   329	                    node,
   330	                    target_url: url.to_string(),
   331	                    status: response.status().as_u16(),
   332	                    latency_ms,
   333	                    success: response.status().is_success(),
   334	                    error: String::new(),
   335	                    placement_status,
   336	                    placement_error,
   337	                },
   338	                Err(error) => HttpObservation {
   339	                    timestamp_utc: timestamp.to_rfc3339(),
   340	                    node,
   341	                    target_url: url.to_string(),
   342	                    status: 0,
   343	                    latency_ms,
   344	                    success: false,
   345	                    error: error.to_string(),
   346	                    placement_status,
   347	                    placement_error,
   348	                },
   349	            }
   350	        } else {
   351	            // Preserve legacy node-selected Nginx measurement behavior.
   352	            let active_node = discover_active_node(
   353	                &config.namespace, &config.pod_label_selector
   354	            ).await;
   355	            let (node, target_url, discovery_error) = match active_node {
   356	                Ok(Some(node)) => match config.service_url_for_node(&node) {
   357	                    Some(url) => (node, url, String::new()),
   358	                    None => (node.clone(), String::new(),
   359	                        format!("no service target configured for node {}", node)),
   360	                },
   361	                Ok(None) => (String::new(), String::new(),
   362	                    "no Running workload pod found".to_string()),
   363	                Err(error) => (String::new(), String::new(),
   364	                    format!("placement lookup failed: {}", error)),
   365	            };
   366	            let start = Instant::now();
   367	            if target_url.is_empty() {
   368	                HttpObservation {
   369	                    timestamp_utc: timestamp.to_rfc3339(),
   370	                    node,
   371	                    target_url,
   372	                    status: 0,
   373	                    latency_ms: 0.0,
   374	                    success: false,
   375	                    error: discovery_error.clone(),
   376	                    placement_status: "UNKNOWN".to_string(),
   377	                    placement_error: discovery_error,
   378	                }
   379	            } else {
   380	                match client.get(&target_url).send().await {
   381	                    Ok(response) => HttpObservation {
   382	                        timestamp_utc: timestamp.to_rfc3339(),
   383	                        node,
