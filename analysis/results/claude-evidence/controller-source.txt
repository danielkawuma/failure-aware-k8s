    65

    74	    fn validate(&self) -> Result<()> {
    75	        if self.target_node.trim().is_empty() {
    76	            bail!("target_node cannot be empty");
    77	        }
    78	
    79	        if self.namespace.trim().is_empty() {
    80	            bail!("namespace cannot be empty");
    81	        }
    82	
    83	        if self.pod_label_selector.trim().is_empty() {
    84	            bail!("pod_label_selector cannot be empty");
    85	        }
    86	
    87	        if self.verification_url.trim().is_empty() {
    88	            bail!("verification_url cannot be empty");
    89	        }
    90	
    91	        if self.verification_attempts == 0 {
    92	            bail!("verification_attempts must be positive");
    93	        }
    94	
    95	        for (name, value) in [
    96	            ("verification_timeout_ms", self.verification_timeout_ms),
    97	            ("kubernetes_api_timeout_ms", self.kubernetes_api_timeout_ms),
    98	            (
    99	                "resource_evaluation_timeout_ms",
   100	                self.resource_evaluation_timeout_ms,
   101	            ),
   102	            ("enforcement_timeout_ms", self.enforcement_timeout_ms),
   103	            ("fencing_timeout_ms", self.fencing_timeout_ms),
   104	        ] {
   105	            if value == 0 {
   106	                bail!("{name} must be greater than zero");
   107	            }
   108	        }
   109	
   110	        if self.resource_evaluation_timeout_ms < self.kubernetes_api_timeout_ms {
   111	            bail!(
   112	                "resource evaluation timeout must be at least \
   113	                 as large as the individual API timeout"
   114	            );
   115	        }
   116	
   117	        if self.duplication_sensitive
   118	            && self.fencing_supported
   119	            && self.fencing_command.trim().is_empty()
   120	        {
   121	            bail!(
   122	                "duplication-sensitive recovery requires a \
   123	                 configured fencing command when fencing \
   124	                 support is enabled"
   125	            );
   126	        }
   127	
   128	        Ok(())
   129	    }

    57	pub async fn run_suspicion_listener(
    58	    client: Client,
    59	    target_node: String,
    60	    trigger_status: String,
    61	    verification_url: String,
    62	    verification_timeout_ms: u64,
    63	    verification_attempts: u32,
    64	    verification_interval_ms: u64,
    65	    workload_awareness: WorkloadAwareness,
    66	    namespace: String,
    67	    pod_label_selector: String,
    68	    recovery_candidate_node: String,
    69	    required_cpu_millicores: u64,
    70	    required_memory_mib: u64,
    71	    required_node_labels: HashMap<String, String>,
    72	    fencing_supported: bool,
    73	    fencing_command: String,
    74	    resource_evaluation_timeout_ms: u64,
    75	    _enforcement_timeout_ms: u64,
    76	) -> Result<()> {
    77	    let nodes: Api<Node> = Api::all(client.clone());
    78	
    79	    let watcher_config =
    80	        watcher::Config::default().fields(&format!("metadata.name={}", target_node));
    81	
    82	    let mut stream = watcher::watcher(nodes, watcher_config).boxed();
    83	
    84	    let mut previous_ready: Option<String> = None;
    85	
    86	    println!(
    87	        "[LISTENER] Watching node={} for Ready={}",
    88	        target_node, trigger_status
    89	    );
    90	
    91	    while let Some(event) = stream.try_next().await? {
    92	        match event {
    93	            Event::Init => {
    94	                println!("[LISTENER] Initializing node watch");
    95	            }
    96	
    97	            Event::InitApply(node) => {
    98	                previous_ready = ready_status(&node);
    99	
   100	                println!(
   101	                    "[INITIAL STATE] node={} | Ready={:?}",
   102	                    target_node, previous_ready
   103	                );
   104	            }
   105	
   106	            Event::InitDone => {
   107	                println!("[LISTENER] Initial node state synchronized");
   108	            }
   109	
   110	            Event::Apply(node) => {
   111	                let current_ready = ready_status(&node).unwrap_or_else(|| "Unknown".to_string());
   112	
   113	                let new_suspicion = current_ready == trigger_status
   114	                    && previous_ready.as_deref() != Some(trigger_status.as_str());
   115	
   116	                if new_suspicion {
   117	                    let suspicion = FailureSuspicion {
   118	                        node: target_node.clone(),
   119	                        detected_at: Utc::now(),
   120	                        previous_ready: previous_ready.clone(),
   121	                        current_ready: current_ready.clone(),
   122	                    };
   123	
   124	                    println!(
   125	                        "[SUSPICION] {} | node={} | Ready={:?} -> {}",
   126	                        suspicion.detected_at.to_rfc3339(),
   127	                        suspicion.node,
   128	                        suspicion.previous_ready,
   129	                        suspicion.current_ready
   130	                    );

   133	                        &verification_url,
   134	                        verification_timeout_ms,
   135	                        verification_attempts,
   136	                        verification_interval_ms,
   137	                    )

   140	                    let operational_state = awareness::classify_operational_state(&evidence);
   141	
   142	                    println!(
   143	                        "[VERIFICATION] node={} | state={:?} | attempts={} | successes={} | failures={}",

   145	                        operational_state,
   146	                        evidence.attempts,
   147	                        evidence.successes,
   148	                        evidence.failures

   154	                                "[SHORT CIRCUIT] node={} | candidate_evaluation=skipped | workload_discovery=skipped | reason=reachable",

   156	                            );

   159	                                "workload remains reachable outside the control-plane path; unnecessary recovery is avoided",

   161	                        }

   350	                                    Utc::now().to_rfc3339(),
   351	                                    target_node,
   352	                                    error
   353	                                );
   354	                            }
   355	                        }
   356	                    }
   357	                }
   358	
   359	                if previous_ready.as_deref() == Some(trigger_status.as_str())
   360	                    && current_ready == "True"

   171	async fn probe_from_worker4(
   172	    target: &SshPodTarget,
   173	    pod: &OriginalPod,
   174	    timeout_ms: u64,
   175	) -> VerificationResult {
   176	    let checked_at = Utc::now();
   177	    let started = Instant::now();
   178	
   179	    let user = env::var("FAILURE_AWARE_VERIFIER_SSH_USER")
   180	        .unwrap_or_else(|_| "ubuntu".to_string());
   181	
   182	    if user.is_empty()
   183	        || !user.chars().all(|c| {
   184	            c.is_ascii_alphanumeric() || c == '_' || c == '-'
   185	        })
   186	    {
   187	        return indeterminate_result(
   188	            "Invalid verifier SSH user".to_string()
   189	        );
   190	    }
   191	
   192	    let home = env::var("HOME").unwrap_or_default();
   193	
   194	    let key = env::var("FAILURE_AWARE_VERIFIER_SSH_KEY")
   195	        .unwrap_or_else(|_| {
   196	            format!("{home}/.ssh/k8s-aware")
   197	        });

   199	    let remote = format!(
   200	        "{}@{}",
   201	        user,
   202	        target.verifier_host

   206	    // The remote command never follows the gateway or Service.
   207	    //
   208	    // Exit status 0 from SSH means the remote probe executed.
   209	    // Its output distinguishes HTTP failure from verifier failure.
   210	    let remote_command = format!(
   211	        "code=$(curl -sS --noproxy '*' \
   212	         --connect-timeout 2 --max-time 3 \
   213	         -o /dev/null -w '%{{http_code}}' \
   214	         'http://{}:8080/openmrs/health/started' \
   215	         2>/dev/null); \
   216	         rc=$?; \
   217	         printf 'PROBE_RESULT:%s:%s\\n' \"$rc\" \"$code\"",
   218	        pod.ip

   222	
   223	    command
   224	        .arg("-i")
   225	        .arg(&key)
   226	        .arg("-o")
   227	        .arg("BatchMode=yes")
   228	        .arg("-o")
   229	        .arg("StrictHostKeyChecking=yes")
   230	        .arg("-o")
   231	        .arg("ConnectTimeout=3")
   232	        .arg("-o")
   233	        .arg("ServerAliveInterval=2")
   234	        .arg("-o")
   235	        .arg("ServerAliveCountMax=1")
   236	        .arg(remote)
   237	        .arg(remote_command)
   238	        .kill_on_drop(true);

   241	        timeout_ms.saturating_add(5000)

   245	        Ok(Ok(output)) => output,

   248	                "SSH execution failed: {error}"

   251	        Err(_) => {
   252	            return indeterminate_result(
   253	                "SSH verifier timed out".to_string()
   254	            );
   255	        }
   256	    };
   257	
   258	    let latency_ms = started.elapsed().as_secs_f64() * 1000.0;
   259	
   260	    if !output.status.success() {
   261	        return indeterminate_result(format!(
   262	            "SSH verifier failed: {}",
   263	            String::from_utf8_lossy(&output.stderr)
   264	        ));
   265	    }
   266	
   267	    let stdout = String::from_utf8_lossy(&output.stdout);
   268	
   269	    let line = match stdout
   270	        .lines()
   271	        .find(|line| line.starts_with("PROBE_RESULT:"))
   272	    {
   273	        Some(line) => line,
   274	        None => {
   275	            return indeterminate_result(
   276	                "Verifier returned no valid probe result".to_string()
   277	            );
   278	        }
   279	    };
   280

   282	
   283	    if fields.len() != 3 {
   284	        return indeterminate_result(
   285	            "Malformed verifier response".to_string()
   286	        );
   287	    }
   288	
   289	    let curl_rc: i32 = match fields[1].parse() {
   290	        Ok(value) => value,
   291	        Err(_) => {
   292	            return indeterminate_result(
   293	                "Invalid remote curl exit code".to_string()
   294	            );
   295	        }
   296	    };
   297	
   298	    let status: u16 = match fields[2].parse() {
   299	        Ok(value) => value,
   300	        Err(_) => {
   301	            return indeterminate_result(
   302	                "Invalid remote HTTP status".to_string()
   303	            );
   304	        }
   305	    };
   306

   308	
   309	    VerificationResult {
   310	        checked_at,
   311	        reachable,
   312	        http_status: if status == 0 {
   313	            None
   314	        } else {
   315	            Some(status)
   316	        },
   317	        latency_ms,
   318	        error: if reachable {
   319	            None
   320	        } else {
   321	            Some(format!(
   322	                "Original backend probe failed: curl_rc={curl_rc}, HTTP={status}"
   323	            ))
   324	        },
   325	    }

   366	pub async fn verify_http_bounded(
   367	    url: &str,
   368	    timeout_ms: u64,
   369	    attempts: u32,
   370	    interval_ms: u64,
   371	) -> Result<VerificationEvidence> {
   372	    let mut results = Vec::new();
   373	    let mut successes = 0;
   374	    let mut failures = 0;
   375	    let mut indeterminate = false;
   376	
   377	    if attempts == 0 {
   378	        return Err(anyhow!("Verification attempts must be positive"));
   379	    }
   380	
   381	    // Resolve the original Pod once for the complete sequence.
   382	    // Never select a replacement between probe attempts.
   383	    let ssh_context = if url.starts_with("ssh-pod://") {
   384	        match parse_ssh_target(url) {
   385	            Ok(target) => {
   386	                match discover_original(&target, timeout_ms).await {
   387	                    Ok(pod) => {
   388	                        println!(
   389	                            "[VERIFY TARGET] pod={} uid={} node={} ip={} verifier={}",
   390	                            pod.name,
   391	                            pod.uid,
   392	                            target.node,
   393	                            pod.ip,
   394	                            target.verifier_host
   395	                        );
   396	
   397	                        Some((target, pod))
   398	                    }
   399	
   400	                    Err(error) => {
   401	                        println!(
   402	                            "[VERIFY INDETERMINATE] Original Pod discovery: {}",
   403	                            error
   404	                        );
   405	
   406	                        return Ok(VerificationEvidence {
   407	                            attempts: 0,
   408	                            successes: 0,
   409	                            failures: 0,
   410	                            results: vec![
   411	                                indeterminate_result(error.to_string())
   412	                            ],
   413	                            indeterminate: true,
   414	                        });
   415	                    }
   416	                }
   417	            }
   418	
   419	            Err(error) => {
   420	                return Ok(VerificationEvidence {
   421	                    attempts: 0,
   422	                    successes: 0,
   423	                    failures: 0,
   424	                    results: vec![
   425	                        indeterminate_result(error.to_string())
   426	                    ],
   427	                    indeterminate: true,
   428	                });
   429	            }
   430	        }
   431	    } else {
   432	        None
   433	    };
   434	
   435	    for attempt in 1..=attempts {
   436	        let result = match &ssh_context {
   437	            Some((target, pod)) => {
   438	                probe_from_worker4(target, pod, timeout_ms).await
   439	            }
   440	            None => verify_http(url, timeout_ms).await?,
   441	        };
   442	
   443	        if is_indeterminate(&result) {
   444	            indeterminate = true;
   445	        } else if result.reachable {
   446	            successes += 1;
   447	        } else {
   448	            failures += 1;
   449	        }
   450	
   451	        println!(
   452	            "[VERIFY] attempt={}/{} | reachable={} | HTTP={:?} | latency={:.3} ms | error={:?}",
   453	            attempt,
   454	            attempts,
   455	            result.reachable,
   456	            result.http_status,
   457	            result.latency_ms,
   458	            result.error
   459	        );
   460

   462	
   463	        if attempt < attempts {
   464	            sleep(Duration::from_millis(interval_ms)).await;
   465	        }
   466	    }
   467	
   468	    Ok(VerificationEvidence {
   469	        attempts,
   470	        successes,
   471	        failures,
   472	        results,
   473	        indeterminate,
   474	    })
   475	}

    22	    evidence: &VerificationEvidence,
    23	) -> OperationalState {
    24	    // Failure of the independent verification mechanism
    25	    // is not evidence that the original workload failed.
    26	    //
    27	    // Do not authorize recovery on incomplete or
    28	    // unverifiable evidence.
    29	    if evidence.indeterminate || evidence.attempts == 0 {

    31	    }
    32	
    33	    if evidence.successes > evidence.failures {
    34	        OperationalState::ReachableOutsideControlPlane
    35	    } else if evidence.successes == 0
    36	        && evidence.failures > 0
    37	    {
    38	        OperationalState::Unreachable
    39	    } else {
    40	        OperationalState::Indeterminate
    41	    }
    42	}

   116	    api_timeout_ms: u64,
   117	) -> Result<WorkloadAwareness> {

   120	    }

   123	
   124	    let params = ListParams::default().labels(pod_label_selector);
   125	
   126	    let response = timeout(Duration::from_millis(api_timeout_ms), pods.list(&params))
   127	        .await
   128	        .map_err(|_| {
   129	            anyhow!(
   130	                "Workload label discovery timed out after {} ms",
   131	                api_timeout_ms
   132	            )
   133	        })??;
   134	
   135	    let sensitivity = resolve_pod_sensitivity(
   136	        &response.items,
   137	        target_node,
   138	        configured.duplication_sensitive,
   139	    )?;
   140	
   141	    println!(
   142	        "[WORKLOAD] name={} | target_node={} | duplication_sensitive={}",
   143	        configured.name, target_node, sensitivity
   144	    );
   145	
   146	    Ok(configured.with_duplication_sensitivity(sensitivity))
   147	}

   161	fn parse_cpu_millicores(value: &str) -> Result<u64> {
   162	    if let Some(nano) = value.strip_suffix('n') {
   163	        let nano: u64 = nano.parse()?;
   164	        return Ok(nano.saturating_add(999_999) / 1_000_000);
   165	    }
   166	
   167	    if let Some(micro) = value.strip_suffix('u') {
   168	        let micro: u64 = micro.parse()?;
   169	        return Ok(micro.saturating_add(999) / 1_000);
   170	    }
   171	
   172	    if let Some(milli) = value.strip_suffix('m') {
   173	        return Ok(milli.parse()?);
   174	    }
   175	
   176	    let cores: f64 = value.parse()?;
   177	
   178	    if !cores.is_finite() || cores < 0.0 {
   179	        return Err(anyhow!("Invalid CPU quantity: {}", value));
   180	    }
   181	
   182	    Ok((cores * 1000.0).ceil() as u64)
   183	}

   185	fn parse_memory_mib(value: &str) -> Result<u64> {
   186	    let units = [
   187	        ("Ki", 1024.0),
   188	        ("Mi", 1024.0_f64.powi(2)),
   189	        ("Gi", 1024.0_f64.powi(3)),
   190	        ("Ti", 1024.0_f64.powi(4)),
   191	        ("Pi", 1024.0_f64.powi(5)),
   192	        ("Ei", 1024.0_f64.powi(6)),
   193	        ("K", 1000.0),
   194	        ("M", 1000.0_f64.powi(2)),
   195	        ("G", 1000.0_f64.powi(3)),
   196	        ("T", 1000.0_f64.powi(4)),
   197	        ("P", 1000.0_f64.powi(5)),
   198	        ("E", 1000.0_f64.powi(6)),
   199	    ];
   200	
   201	    for (suffix, multiplier) in units {
   202	        if let Some(number) = value.strip_suffix(suffix) {
   203	            let amount: f64 = number.parse()?;
   204	
   205	            if !amount.is_finite() || amount < 0.0 {
   206	                return Err(anyhow!("Invalid memory quantity: {}", value));
   207	            }
   208	
   209	            return Ok((amount * multiplier / 1024.0 / 1024.0).ceil() as u64);
   210	        }
   211	    }
   212	
   213	    let bytes: f64 = value.parse()?;
   214	
   215	    if !bytes.is_finite() || bytes < 0.0 {
   216	        return Err(anyhow!("Invalid memory quantity: {}", value));
   217	    }
   218	
   219	    Ok((bytes / 1024.0 / 1024.0).ceil() as u64)
   220	}

   242	fn container_requests(pod: &Pod) -> Result<(u64, u64)> {
   243	    let Some(spec) = pod.spec.as_ref() else {
   244	        return Ok((0, 0));
   245	    };
   246	
   247	    let mut app_cpu = 0u64;
   248	    let mut app_memory = 0u64;
   249	
   250	    for container in &spec.containers {
   251	        if let Some(requests) = container
   252	            .resources
   253	            .as_ref()
   254	            .and_then(|resources| resources.requests.as_ref())
   255	        {
   256	            if let Some(cpu) = requests.get("cpu") {
   257	                app_cpu = app_cpu.saturating_add(parse_cpu_millicores(&cpu.0)?);
   258	            }
   259	
   260	            if let Some(memory) = requests.get("memory") {
   261	                app_memory = app_memory.saturating_add(parse_memory_mib(&memory.0)?);
   262	            }
   263	        }
   264	    }
   265	
   266	    let mut max_init_cpu = 0u64;
   267	    let mut max_init_memory = 0u64;
   268	
   269	    if let Some(init_containers) = spec.init_containers.as_ref() {
   270	        for container in init_containers {
   271	            let mut cpu = 0u64;
   272	            let mut memory = 0u64;
   273	
   274	            if let Some(requests) = container
   275	                .resources
   276	                .as_ref()
   277	                .and_then(|resources| resources.requests.as_ref())
   278	            {
   279	                if let Some(value) = requests.get("cpu") {
   280	                    cpu = parse_cpu_millicores(&value.0)?;
   281	                }
   282	
   283	                if let Some(value) = requests.get("memory") {
   284	                    memory = parse_memory_mib(&value.0)?;
   285	                }
   286	            }
   287	
   288	            max_init_cpu = max_init_cpu.max(cpu);
   289	            max_init_memory = max_init_memory.max(memory);
   290	        }
   291	    }
   292	
   293	    let mut cpu = app_cpu.max(max_init_cpu);
   294	    let mut memory = app_memory.max(max_init_memory);
   295	
   296	    if let Some(overhead) = spec.overhead.as_ref() {
   297	        if let Some(value) = overhead.get("cpu") {
   298	            cpu = cpu.saturating_add(parse_cpu_millicores(&value.0)?);
   299	        }
   300	
   301	        if let Some(value) = overhead.get("memory") {
   302	            memory = memory.saturating_add(parse_memory_mib(&value.0)?);
   303	        }
   304	    }
   305	
   306	    Ok((cpu, memory))
   307	}

   345	fn reliability_score(history: &NodeHistory) -> f64 {
   346	    let uptime = history.uptime_ratio.clamp(0.0, 1.0);
   347	
   348	    let recent_failure_health = 1.0 / (1.0 + history.recent_failures as f64);
   349	
   350	    let recovery_total = history.successful_recoveries + history.failed_recoveries;
   351	
   352	    let recovery_stability = if recovery_total == 0 {
   353	        0.5
   354	    } else {
   355	        history.successful_recoveries as f64 / recovery_total as f64
   356	    };
   357	
   358	    0.60 * uptime + 0.25 * recent_failure_health + 0.15 * recovery_stability
   359	}

   457	    nodes: &[Node],
   458	    history: &mut HashMap<String, NodeHistory>,
   459	    state: &mut ObservationState,

   492	fn recompute_uptime_ratios(history: &mut HashMap<String, NodeHistory>) -> Result<()> {
   493	    let path = observations_path();
   494	
   495	    if !path.exists() {
   496	        return Ok(());
   497	    }
   498	
   499	    let file = fs::File::open(path)?;
   500	    let reader = BufReader::new(file);
   501	
   502	    let mut by_node: HashMap<String, Vec<NodeObservation>> = HashMap::new();
   503	
   504	    for line in reader.lines() {
   505	        let line = line?;
   506	
   507	        if line.trim().is_empty() {
   508	            continue;
   509	        }
   510	
   511	        let observation: NodeObservation = serde_json::from_str(&line)?;
   512	
   513	        by_node
   514	            .entry(observation.node.clone())
   515	            .or_default()
   516	            .push(observation);
   517	    }
   518	
   519	    for (node, mut observations) in by_node {
   520	        observations.sort_by_key(|o| o.timestamp_utc);
   521	
   522	        let mut ready_ms: i64 = 0;
   523	        let mut known_ms: i64 = 0;
   524	
   525	        for pair in observations.windows(2) {
   526	            let first = &pair[0];
   527	            let second = &pair[1];
   528	
   529	            let elapsed = (second.timestamp_utc - first.timestamp_utc).num_milliseconds();
   530	
   531	            if elapsed <= 0 || elapsed > 60_000 {
   532	                continue;
   533	            }
   534	
   535	            match first.ready_status.as_str() {
   536	                "True" => {
   537	                    ready_ms += elapsed;
   538	                    known_ms += elapsed;
   539	                }
   540	
   541	                "False" => {
   542	                    known_ms += elapsed;
   543	                }
   544	
   545	                _ => {}
   546	            }
   547	        }
   548	
   549	        if known_ms > 0 {
   550	            let entry = history.entry(node).or_default();
   551	
   552	            entry.uptime_ratio = ready_ms as f64 / known_ms as f64;
   553	        }
   554	    }
   555	
   556	    Ok(())
   557	}

   615	fn evaluate_one_candidate(
   616	    node: &Node,
   617	    required_cpu_millicores: u64,
   618	    required_memory_mib: u64,
   619	    required_labels: &HashMap<String, String>,
   620	    history: &HashMap<String, NodeHistory>,
   621	    usage: &HashMap<String, (u64, u64)>,
   622	) -> Result<CandidateEvaluation> {
   623	    let node_name = node
   624	        .metadata
   625	        .name
   626	        .clone()
   627	        .ok_or_else(|| anyhow!("Candidate node has no name"))?;
   628	
   629	    let ready = node_ready(node);
   630	    let schedulable = node_schedulable(node);
   631	    let labels_ok = labels_match(node, required_labels);
   632	
   633	    let (allocatable_cpu, allocatable_memory) = node_allocatable(node)?;
   634	
   635	    let (requested_cpu, requested_memory) = usage.get(&node_name).copied().unwrap_or((0, 0));
   636	
   637	    let remaining_cpu = allocatable_cpu.saturating_sub(requested_cpu);
   638	
   639	    let remaining_memory = allocatable_memory.saturating_sub(requested_memory);
   640	
   641	    let score = reliability_score(history.get(&node_name).unwrap_or(&NodeHistory::default()));
   642	
   643	    let sufficient =
   644	        remaining_cpu >= required_cpu_millicores && remaining_memory >= required_memory_mib;
   645	
   646	    let eligible = ready && schedulable && labels_ok && sufficient;
   647	
   648	    Ok(CandidateEvaluation {
   649	        node_name,
   650	        ready,
   651	        schedulable,
   652	        labels_match: labels_ok,
   653	
   654	        allocatable_cpu_millicores: allocatable_cpu,
   655	        requested_cpu_millicores: requested_cpu,
   656	        remaining_cpu_millicores: remaining_cpu,
   657	
   658	        allocatable_memory_mib: allocatable_memory,
   659	        requested_memory_mib: requested_memory,
   660	        remaining_memory_mib: remaining_memory,
   661	
   662	        reliability_score: score,
   663	        eligible,
   664	    })
   665	}

    17	pub fn evaluate_recovery_safety(
    18	    workload: &WorkloadAwareness,
    19	    fencing_supported: bool,
    20	) -> RecoverySafetyAwareness {
    21	    /*
    22	     * Preserve the existing safety policy:
    23	     *
    24	     * Duplication sensitivity determines whether fencing
    25	     * is required before a replacement may be released.
    26	     *
    27	     * Workload reachability alone does not prove that
    28	     * the original writer has lost authority.
    29	     */
    30	
    31	    if !workload.duplication_sensitive {
    32	        return RecoverySafetyAwareness {
    33	            state: RecoverySafetyState::Safe,
    34	            fencing_required: false,
    35	            fencing_supported,
    36	        };
    37	    }
    38	
    39	    /*
    40	     * Fencing support means a fencing mechanism is
    41	     * configured, not that fencing has succeeded.
    42	     *
    43	     * Actual fencing must be executed and verified
    44	     * by enforcement.rs before releasing recovery.
    45	     */
    46	    if fencing_supported {
    47	        return RecoverySafetyAwareness {
    48	            state: RecoverySafetyState::FencingRequired,
    49	            fencing_required: true,
    50	            fencing_supported: true,
    51	        };
    52	    }
    53	
    54	    /*
    55	     * Fail closed when the workload is duplication
    56	     * sensitive and no fencing mechanism is available.
    57	     */
    58	    RecoverySafetyAwareness {
    59	        state: RecoverySafetyState::Unsafe,
    60	        fencing_required: true,
    61	        fencing_supported: false,
    62	    }
    63	}

    87	    let mut command = Command::new("sh");
    88	
    89	    command
    90	        .arg("-c")
    91	        .arg(fencing_command)
    92	        .stdin(Stdio::null())
    93	        .stdout(Stdio::piped())
    94	        .stderr(Stdio::piped())
    95	        .kill_on_drop(true);
    96	
    97	    let child = match command.spawn() {
    98	        Ok(child) => child,
    99	        Err(error) => {
   100	            return Ok(failed(
   101	                fencing_command,
   102	                format!("failed to start fencing command: {}", error),
   103	            ));
   104	        }
   105	    };
   106	
   107	    let result = timeout(Duration::from_millis(timeout_ms), child.wait_with_output()).await;
   108	
   109	    match result {
   110	        Ok(Ok(output)) => {
   111	            if output.status.success() {
   112	                let stdout = String::from_utf8_lossy(&output.stdout).trim().to_string();
   113	
   114	                println!(
   115	                    "[FENCING RESULT] {} | state=Succeeded",
   116	                    Utc::now().to_rfc3339()
   117	                );
   118	
   119	                Ok(FencingResult {
   120	                    state: FencingState::Succeeded,
   121	                    command: fencing_command.to_string(),
   122	                    message: if stdout.is_empty() {
   123	                        "fencing command completed successfully".to_string()
   124	                    } else {
   125	                        stdout
   126	                    },
   127	                })
   128	            } else {
   129	                let stderr = String::from_utf8_lossy(&output.stderr).trim().to_string();
   130	
   131	                let message = format!(
   132	                    "fencing command exited with status {}: {}",
   133	                    output.status, stderr
   134	                );
   135	
   136	                println!(
   137	                    "[FENCING RESULT] {} | state=Failed | reason={}",
   138	                    Utc::now().to_rfc3339(),
   139	                    message
   140	                );
   141	
   142	                Ok(failed(fencing_command, message))
   143	            }
   144	        }
   145	
   146	        Ok(Err(error)) => {
   147	            let message = format!("fencing execution failed: {}", error);
   148	
   149	            println!(
   150	                "[FENCING RESULT] {} | state=Failed | reason={}",
   151	                Utc::now().to_rfc3339(),
   152	                message
   153	            );
   154	
   155	            Ok(failed(fencing_command, message))
   156	        }
   157	
   158	        Err(_) => {
   159	            let message = format!(
   160	                "fencing exceeded {} ms; outcome is unverified and recovery must remain blocked",
   161	                timeout_ms
   162	            );
   163	
   164	            println!(
   165	                "[FENCING TIMEOUT] {} | timeout_ms={} | recovery_authorized=false",
   166	                Utc::now().to_rfc3339(),
   167	                timeout_ms
   168	            );
   169	
   170	            Ok(failed(fencing_command, message))
   171	        }
   172	    }
   173	}

   270	async fn wait_for_replacement(
   271	    client: Client,
   272	    namespace: &str,
   273	    selector: &str,
   274	    original_pod: &str,
   275	    candidate: Option<&str>,
   276	) -> Result<String> {
   277	    let pods: Api<Pod> = Api::namespaced(client, namespace);
   278	
   279	    let deadline = tokio::time::Instant::now() + Duration::from_millis(recovery_timeout_ms());
   280	
   281	    loop {
   282	        if tokio::time::Instant::now() >= deadline {
   283	            return Err(anyhow!("Replacement readiness verification timed out"));
   284	        }
   285	
   286	        let list = bounded_api("list replacement Pods", async {
   287	            Ok(pods.list(&ListParams::default().labels(selector)).await?)
   288	        })
   289	        .await?;
   290	
   291	        for pod in list.items {
   292	            let name = pod.metadata.name.as_deref().unwrap_or("");
   293	
   294	            if name == original_pod {
   295	                continue;
   296	            }
   297	
   298	            if pod.metadata.deletion_timestamp.is_some() {
   299	                continue;
   300	            }
   301	
   302	            let Some(spec) = pod.spec.as_ref() else {
   303	                continue;
   304	            };
   305	
   306	            if let Some(node) = candidate {
   307	                if spec.node_name.as_deref() != Some(node) {
   308	                    continue;
   309	                }
   310	            }
   311	
   312	            let running = pod.status.as_ref().and_then(|s| s.phase.as_deref()) == Some("Running");
   313	
   314	            let ready = pod
   315	                .status
   316	                .as_ref()
   317	                .and_then(|s| s.conditions.as_ref())
   318	                .map(|conditions| {
   319	                    conditions
   320	                        .iter()
   321	                        .any(|condition| condition.type_ == "Ready" && condition.status == "True")
   322	                })
   323	                .unwrap_or(false);
   324	
   325	            if running && ready {
   326	                return Ok(name.to_string());
   327	            }
   328	        }
   329	
   330	        sleep(Duration::from_secs(2)).await;
   331	    }
   332	}

   372	pub async fn reconcile_placement_constraint(

   396	                .spec

   398	                .ok_or_else(|| anyhow!("Missing Pod template"))?;

   402	                workload.metadata.annotations.clone().unwrap_or_default(),

   405	                        .match_labels

   415	            )
   416	        }

   418	        "StatefulSet" => {

   420

   424	            .await?;

   426	            let spec = workload
   427	                .spec
   428	                .as_ref()
   429	                .ok_or_else(|| anyhow!("Missing StatefulSet spec"))?;

   431	            let pod_spec = spec
   432	                .template
   433	                .spec
   434	                .as_ref()
   435	                .ok_or_else(|| anyhow!("Missing Pod template"))?;
   436	
   437	            (
   438	                spec.replicas.unwrap_or(1),

   441	                    spec.selector
   442	                        .match_labels

   457	        }
   458	    };

   461	        return Ok(false);
   462	    }
   463	
   464	    if annotations.get(CLEANUP_KEY).map(String::as_str) != Some("pending") {
   465	        return Err(anyhow!("Placement cleanup is not marked pending"));

   492	        Ok(pods.list(&ListParams::default().labels(&selector)).await?)

   506	            "annotations": {

   511	            }
   512	        },
   513	        "spec": {
   514	            "template": {
   515	                "spec": {
   516	                    "nodeSelector": {
   517	                        HOSTNAME_KEY: null
   518	                    }
   519	                }
   520	            }
   521	        }
   522	    });
   523	
   524	    match kind {
   525	        "Deployment" => {
   526	            let api: Api<Deployment> = Api::namespaced(client, namespace);
   527	
   528	            bounded_api("cleanup Deployment placement", async {
   529	                Ok(api
   530	                    .patch(name, &PatchParams::default(), &Patch::Merge(&patch))

   543	            })
   544	            .await?;
   545	        }

   548	    }
   549	
   550	    println!(
   551	        "[PLACEMENT CLEANUP] {} | workload={}/{} | constraint_removed=true",
   552	        Utc::now().to_rfc3339(),
   553	        kind,
   554	        name
   555	    );
   556	
   557	    Ok(true)
   558	}

   572	            enforced: true,
   573	            message: "Existing workload preserved".to_string(),
   574	        });
   575	    }
   576	
   577	    if decision.fencing_required {
   578	        let result = fencing::execute_fence_bounded(
   579	            true,
   580	            fencing_supported,
   581	            fencing_command,
   582	            fencing_timeout_ms(),
   583	        )
   584	        .await?;
   585	
   586	        if result.state != FencingState::Succeeded {
   587	            return Ok(blocked(
   588	                EnforcementAction::RecoveryBlockedFencingFailed,
   589	                format!("Fencing failed: {}", result.message),
   590	            ));
   591	        }
   592	
   593	        println!(
   594	            "[FENCING VERIFIED] {} | node={}",
   595	            Utc::now().to_rfc3339(),
   596	            target_node
   597	        );
   598	    }
   599	
   600	    let pods: Api<Pod> = Api::namespaced(client.clone(), namespace);
   601	
   602	    let list = bounded_api("list target Pods", async {
   603	        Ok(pods
   604	            .list(&ListParams::default().labels(pod_label_selector))
   605	            .await?)
   606	    })
   607	    .await?;
   608	
   609	    let target_pod = list
   610	        .items
   611	        .into_iter()
   612	        .find(|pod| pod.spec.as_ref().and_then(|s| s.node_name.as_deref()) == Some(target_node));
   613	
   614	    let Some(pod) = target_pod else {
   615	        return Ok(blocked(
   616	            EnforcementAction::RecoveryReleased,
   617	            "No matching target Pod found".to_string(),
   618	        ));
   619	    };
   620	
   621	    let pod_name = pod
   622	        .metadata
   623	        .name
   624	        .clone()
   625	        .ok_or_else(|| anyhow!("Pod has no name"))?;
   626	
   627	    if let Some(candidate) = decision.selected_candidate.as_deref() {
   628	        if candidate == target_node {
   629	            return Ok(blocked(
   630	                EnforcementAction::RecoveryBlockedPlacementConstraintFailed,
   631	                "Recovery candidate equals target node".to_string(),
   632	            ));
   633	        }
   634	
   635	        if let Err(error) =
   636	            constrain_replacement_to_candidate(client.clone(), namespace, &pod, candidate).await
   637	        {
   638	            return Ok(blocked(
   639	                EnforcementAction::RecoveryBlockedPlacementConstraintFailed,
   640	                error.to_string(),
   641	            ));
   642	        }
   643	    }
   644	
   645	    bounded_api("delete preserved Pod", async {
   646	        Ok(pods
   647	            .delete(
   648	                &pod_name,
   649	                &DeleteParams {
   650	                    grace_period_seconds: Some(0),
   651	                    ..Default::default()
   652	                },
   653	            )
   654	            .await?)
   655	    })
   656	    .await?;
   657	
   658	    let replacement = wait_for_replacement(
   659	        client,
   660	        namespace,
   661	        pod_label_selector,
   662	        &pod_name,
   663	        decision.selected_candidate.as_deref(),
   664	    )
   665	    .await;
   666	
   667	    let replacement_name = match replacement {
   668	        Ok(name) => name,
   669	        Err(error) => {
   670	            println!("[RECOVERY UNVERIFIED] {}", error);
   671	
   672	            return Ok(blocked(
   673	                EnforcementAction::RecoveryReleased,
   674	                format!("Recovery initiated but not verified: {}", error),
   675	            ));
   676	        }
   677	    };
   678	
   679	    if let Err(error) = verify_application().await {
   680	        println!("[APPLICATION VERIFICATION FAILED] {}", error);
   681	
   682	        return Ok(blocked(
   683	            EnforcementAction::RecoveryReleased,
   684	            format!(
   685	                "Replacement {} is Ready but application verification failed: {}",
   686	                replacement_name, error
   687	            ),
   688	        ));
   689	    }
   690	
   691	    // Optional post-recovery reconciliation. Legacy experiments remain unchanged
   692	    // unless FAILURE_AWARE_RECONCILE_COMMAND is explicitly configured.
   693	    if decision.fencing_required {
   694	        if let Ok(reconcile_command) = env::var("FAILURE_AWARE_RECONCILE_COMMAND") {
   695	            if !reconcile_command.trim().is_empty() {
   696	                println!("[RECONCILIATION] starting post-recovery safety checks");
   697	                let outcome = timeout(
   698	                    Duration::from_millis(configured_timeout(
   699	                        "FAILURE_AWARE_RECONCILE_TIMEOUT_MS",
   700	                        180_000,
   701	                    )),
   702	                    tokio::process::Command::new("sh")
   703	                        .arg("-c")
   704	                        .arg(&reconcile_command)
   705	                        .kill_on_drop(true)
   706	                        .output(),
   707	                )
   708	                .await;
   709	                match outcome {
   710	                    Ok(Ok(output)) if output.status.success() => {
   711	                        println!(
   712	                            "[RECONCILIATION VERIFIED] {}",
   713	                            String::from_utf8_lossy(&output.stdout).trim()
   714	                        );
   715	                    }
   716	                    Ok(Ok(output)) => {
   717	                        eprintln!(
   718	                            "[RECONCILIATION PENDING] exit={} stderr={}",
   719	                            output.status,
   720	                            String::from_utf8_lossy(&output.stderr).trim()
   721	                        );
   722	                    }
   723	                    Ok(Err(error)) => {
   724	                        eprintln!("[RECONCILIATION PENDING] execution error: {}", error)
   725	                    }
   726	                    Err(_) => {
   727	                        eprintln!("[RECONCILIATION PENDING] timed out; fence must remain active")
   728	                    }
   729	                }
   730	            }
   731	        }
   732	    }
   733	
   734	    println!(
   735	        "[RECOVERY VERIFIED] {} | replacement={}",
   736	        Utc::now().to_rfc3339(),
   737	        replacement_name
   738	    );
   739	
   740	    if let Some(candidate) = decision.selected_candidate.as_deref() {
   741	        if let Err(error) = resource::record_recovery_outcome(candidate, true) {
   742	            println!(
   743	                "[HISTORY ERROR] Recovery succeeded but history update failed: {}",
   744	                error
   745	            );
   746	        }
   747	    }
   748	
   749	    Ok(EnforcementResult {
   750	        action: EnforcementAction::RecoveryReleased,
   751	        enforced: true,
   752	        message: format!(
   753	            "Replacement {} verified Ready and application reachable; placement cleanup pending",
   754	            replacement_name
   755	        ),
   756	    })
   757	}
   758	
   759	pub fn log_enforcement(node: &str, result: &EnforcementResult) {
   760	    println!(
   761	        "[ENFORCEMENT] {} | node={} | action={:?} | enforced={} | message={}",
   762	        Utc::now().to_rfc3339(),
   763	        node,
   764	        result.action,
   765	        result.enforced,
   766	        result.message
   767	    );
   768	}
